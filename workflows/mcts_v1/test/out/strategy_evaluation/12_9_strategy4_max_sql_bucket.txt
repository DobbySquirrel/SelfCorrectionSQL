WITH c_cols AS ( SELECT ym.Date, ym.Consumption FROM yearmonth AS ym INNER JOIN customers AS c ON ym.CustomerID = c.CustomerID INNER JOIN transactions_1k AS t ON ym.CustomerID = t.CustomerID WHERE t.Price IS NOT NULL AND t.Amount IS NOT NULL ), c_rows AS ( SELECT Date, SUM(Consumption) AS TotalConsumption FROM c_cols GROUP BY Date ), t_rows AS ( SELECT Date, TotalConsumption FROM c_rows ORDER BY TotalConsumption DESC LIMIT 1 ) SELECT SUBSTR(Date, 1, 4) AS answer FROM t_rows;
