WITH c_cols AS ( SELECT t.`GasStationID`, SUM(t.`Amount`) AS TotalRevenue FROM transactions_1k t GROUP BY t.`GasStationID` ), c_rows AS ( SELECT GasStationID, TotalRevenue FROM c_cols ORDER BY TotalRevenue DESC LIMIT 1 ), t_join AS ( SELECT g.`GasStationID`, g.`ChainID`, g.`Country`, g.`Segment`, c.TotalRevenue FROM gasstations g JOIN c_rows c ON g.`GasStationID` = c.GasStationID ) SELECT GasStationID, TotalRevenue FROM t_join;
WITH filtered_customers AS ( SELECT DISTINCT t.`CustomerID` FROM transactions_1k t JOIN products p ON t.`ProductID` = p.`ProductID` WHERE p.`ProductID` = 5 AND t.`Date` LIKE '201208%' AND t.`Price` / t.`Amount` > 29.00 ) SELECT y.`CustomerID`, y.`Consumption` FROM yearmonth y JOIN filtered_customers fc ON y.`CustomerID` = fc.`CustomerID` WHERE y.`Date` LIKE '201208%';
