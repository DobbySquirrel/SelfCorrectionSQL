WITH c_cols AS ( SELECT ss.`cds`, ss.`AvgScrRead`, s.`Phone` FROM satscores AS ss INNER JOIN schools AS s ON ss.cds = s.`CDSCode` ), c_rows AS ( SELECT `cds`, `AvgScrRead`, `Phone` FROM c_cols WHERE `AvgScrRead` IS NOT NULL ), t_agg AS ( SELECT `cds`, MIN(`AvgScrRead`) AS min_avg_scr_read FROM c_rows GROUP BY `cds` ), t_final AS ( SELECT t_agg.`cds`, t_agg.min_avg_scr_read, c_rows.`Phone` FROM t_agg INNER JOIN c_rows ON t_agg.`cds` = c_rows.`cds` ORDER BY t_agg.min_avg_scr_read ASC LIMIT 1 ) SELECT `Phone` FROM t_final;
WITH c_cols AS ( SELECT ss.`cds`, ss.`AvgScrMath`, ss.`AvgScrRead`, ss.`AvgScrWrite`, s.`County` FROM satscores AS ss INNER JOIN schools AS s ON ss.`cds` = s.`CDSCode` ), c_rows AS ( SELECT `cds`, `AvgScrMath`, `AvgScrRead`, `AvgScrWrite`, `County` FROM c_cols WHERE `AvgScrMath` IS NOT NULL AND `AvgScrRead` IS NOT NULL AND `AvgScrWrite` IS NOT NULL ), c_agg AS ( SELECT `cds`, AVG(`AvgScrMath` + `AvgScrRead` + `AvgScrWrite`) AS `avg_total_score`, `County` FROM c_rows GROUP BY `cds` ), c_min AS ( SELECT `cds`, `avg_total_score`, `County` FROM c_agg ORDER BY `avg_total_score` ASC LIMIT 1 ), final_answer AS ( SELECT (SELECT `AvgScrMath` FROM c_rows WHERE `cds` = c_min.`cds`) AS `avg_math_score`, c_min.`County` FROM c_min ) SELECT * FROM final_answer;
