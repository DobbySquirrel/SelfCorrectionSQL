WITH c_cols AS ( SELECT `isStorySpotlight`, `isTextless`, `id` FROM cards ), c_rows AS ( SELECT `isStorySpotlight`, `isTextless`, `id` FROM c_cols WHERE `isStorySpotlight` = 1 AND `isTextless` = 0 ), t_count AS ( SELECT COUNT(*) AS story_spotlight_count FROM c_rows ), t_total AS ( SELECT COUNT(*) AS total_count FROM cards ), final_percentage AS ( SELECT (CAST(t_count.story_spotlight_count AS REAL) / t_total.total_count) * 100 AS percentage FROM t_count, t_total ), final_output AS ( SELECT `id` FROM c_rows ) SELECT percentage FROM final_percentage UNION ALL SELECT id FROM final_output;
