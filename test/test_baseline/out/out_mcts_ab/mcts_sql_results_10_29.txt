WITH b_types AS (
    SELECT bond_type, COUNT(*) AS count
    FROM bond
    WHERE molecule_id = 'TR010'
    GROUP BY bond_type
),
max_bond_type AS (
    SELECT bond_type
    FROM b_types
    ORDER BY count DESC
    LIMIT 1
),
is_carcinogenic AS (
    SELECT CASE WHEN EXISTS (
        SELECT 1
        FROM molecule m
        JOIN bond b ON m.molecule_id = b.molecule_id
        WHERE m.label = 'TR010' AND b.bond_type = '-'
    ) THEN 'Yes' ELSE 'No' END AS result
)
SELECT mt.bond_type, ic.result
FROM max_bond_type mt
CROSS JOIN is_carcinogenic ic;

WITH sulfur_atom_ids AS (
    SELECT a.atom_id
    FROM atom a
    WHERE a.element = 's'
),

    SELECT b.bond_id
    FROM bond b
    WHERE b.bond_type = ' = '
),
connected_sulfur_atoms AS (
    SELECT ca1.atom_id AS atom1_id, ca2.atom_id AS atom2_id
    FROM connected ca1
    JOIN connected ca2 ON ca1.bond_id = ca2.bond_id AND ca1.atom_id != ca2.atom_id
    WHERE ca1.atom_id IN (SELECT atom_id FROM sulfur_atom_ids)
      AND ca2.atom_id IN (SELECT atom_id FROM sulfur_atom_ids)
)
SELECT DISTINCT ma.molecule_id, ma.element
FROM molecule m
JOIN atom ma ON m.molecule_id = ma.molecule_id
JOIN connected_sulfur_atoms csa ON ma.atom_id = csa.atom1_id OR ma.atom_id = csa.atom2_id
WHERE ma.element = 's';
