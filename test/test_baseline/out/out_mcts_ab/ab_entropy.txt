    FROM Team
    WHERE team_long_name = 'Ajax'
),
ajax_chance_creation AS (
    SELECT MAX(T2.chanceCreationPassing) AS max_chance_creation,
           T2.chanceCreationPassingClass AS classification
    FROM ajax_teams AS T1
    JOIN Team_Attributes AS T2 ON T1.team_api_id = T2.team_api_id
)
SELECT max_chance_creation, classification FROM ajax_chance_creation;


WITH patient_birth AS (
    SELECT ID
    FROM Patient
    WHERE STRFTIME('%Y', Birthday) = '1937'

high_cholesterol_patients AS (
    SELECT L.ID
    FROM Laboratory L
    JOIN patient_birth PB ON L.ID = PB.ID
    WHERE L.`T-CHO` >= 250
),
filtered_patients AS (
    SELECT P.ID
    FROM Patient P

WITH patient_info AS (
    SELECT ID, SEX
    FROM Patient
    WHERE STRFTIME('%Y', Birthday) = '1980' AND Diagnosis = 'RA'
),
female_patients AS (
    SELECT COUNT(*) AS female_count
    FROM patient_info
    WHERE SEX = 'F'
),
total_patients AS (
    SELECT COUNT(*) AS total_count
    FROM patient_info

WITH p_birthday AS (
    SELECT ID
    FROM Patient
    WHERE Birthday = '1982-04-01'
),
l_alp AS (
    SELECT ID, ALP
    FROM Laboratory
    WHERE ID IN (SELECT ID FROM p_birthday)
),
alp_range AS (
    SELECT ID, ALP
    FROM l_alp
    WHERE ALP < 300
)
SELECT p.ID, p.Description, al.ALP
FROM Patient p
JOIN alp_range al ON p.ID = al.ID;
WITH birth_year AS (
    SELECT ID
    FROM Patient
    WHERE STRFTIME('%Y', Birthday) = '1982'
),
albumin_status AS (
    SELECT L.ID, L.ALB,
           CASE WHEN L.ALB BETWEEN 3.5 AND 5.5 THEN 'Normal' ELSE 'Abnormal' END AS Albumin_Status
    FROM Laboratory AS L
    INNER JOIN birth_year b ON L.ID = b.ID
)
SELECT p.ID, p.Description, a.Albumin_Status
FROM Patient p
INNER JOIN albumin_status a ON p.ID = a.ID
ORDER BY p.ID;
WITH max_bilirubin AS (
    SELECT MAX(`T-BIL`) AS max_value
    FROM Laboratory
),
top_bilirubin_patient AS (
    SELECT p.ID, p.SEX, p.Birthday, l.`T-BIL`
    FROM Patient p
    JOIN Laboratory l ON p.ID = l.ID
    WHERE l.`T-BIL` = (SELECT max_value FROM max_bilirubin)
)
SELECT * FROM top_bilirubin_patient;

    FROM male_patients m
    INNER JOIN wbc_normal w ON m.ID = w.ID

    FROM male_wbc_normal m

    FROM male_fg_abnormal

WITH l_cols AS (
    SELECT ID, IGG
    FROM Laboratory
),
l_rows AS (
    SELECT ID
    FROM l_cols
    WHERE IGG BETWEEN 900 AND 2000
),
p_join AS (
    SELECT e.ID
    FROM Examination e
    INNER JOIN Patient p ON e.ID = p.ID
    WHERE e.ID IN (SELECT ID FROM l_rows) AND p.Admission = '+'
),
p_count AS (
    SELECT COUNT(*) AS answer
    FROM p_join
)
SELECT * FROM p_count;

WITH meeting_expenses AS (
    SELECT SUM(expense.cost) AS total_cost, COUNT(event.event_id) AS total_meetings
    FROM expense
    JOIN budget ON expense.link_to_budget = budget.budget_id
    JOIN event ON budget.link_to_event = event.event_id
    WHERE event.type = 'Meeting'
),
meeting_percentage AS (
    SELECT CAST(total_cost AS REAL) / total_meetings * 100 AS percentage
    FROM meeting_expenses
)
SELECT percentage FROM meeting_percentage;

WITH ym_data AS (
    SELECT CustomerID, SUM(Consumption) AS TotalConsumption
    FROM yearmonth
    WHERE Date BETWEEN '201201' AND '201212'
    GROUP BY CustomerID
)
SELECT c.CustomerID, c.Segment, c.Currency, y.TotalConsumption
FROM ym_data y
JOIN customers c ON y.CustomerID = c.CustomerID
ORDER BY y.TotalConsumption ASC
LIMIT 1;


WITH c_currency AS (
    SELECT CustomerID
    FROM customers
    WHERE Currency = 'Euro'
)

FROM transactions_1k
JOIN c_currency ON transactions_1k.CustomerID = c_currency.CustomerID;
WITH g_station_revenue AS (
    SELECT gs.GasStationID, SUM(t.Amount) AS TotalRevenue
    FROM gasstations gs
    JOIN transactions_1k t ON gs.GasStationID = t.GasStationID
    GROUP BY gs.GasStationID
),
g_station_ranked AS (
    SELECT GasStationID, TotalRevenue,
           ROW_NUMBER() OVER (ORDER BY TotalRevenue DESC) AS rank
    FROM g_station_revenue
),
top_gas_station AS (
    SELECT GasStationID
    FROM g_station_ranked
    WHERE rank = 1
),
top_gas_station_details AS (
    SELECT gs.*
    FROM gasstations gs
    JOIN top_gas_station tg ON gs.GasStationID = tg.GasStationID
)
SELECT * FROM top_gas_station_details;
WITH filtered_transactions AS (
    SELECT CustomerID, Date, Time, ProductID, Price / Amount AS price_per_unit
    FROM transactions_1k
    WHERE ProductID = 5 AND Price / Amount > 29.00
),
customer_consumption AS (
    SELECT ym.CustomerID, ym.Date, ym.Consumption
    FROM yearmonth ym
    INNER JOIN filtered_transactions ft ON ym.CustomerID = ft.CustomerID
    WHERE ym.Date LIKE '201208%'
),
final_status AS (
    SELECT CustomerID, CASE WHEN MAX(Consumption) > 0 THEN 'Active' ELSE 'Inactive' END AS Consumption_Status
    FROM customer_consumption
    GROUP BY CustomerID
)
SELECT * FROM final_status;
