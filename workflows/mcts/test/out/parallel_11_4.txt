WITH m_cols AS ( SELECT first_name, last_name, position, link_to_major FROM member ), m_rows AS ( SELECT first_name, last_name, position, link_to_major FROM m_cols WHERE position = 'Secretary' ), m_joined AS ( SELECT m.first_name, m.last_name, m.position, m.link_to_major, ma.college FROM m_rows m INNER JOIN major ma ON m.link_to_major = ma.major_id ), final_result AS ( SELECT first_name, last_name, college FROM m_joined ) SELECT * FROM final_result;
WITH budget_filter AS ( SELECT budget_id, spent FROM budget WHERE category = 'Speaker Gifts' ), event_details AS ( SELECT e.event_name, b.budget_id FROM event e JOIN budget b ON e.event_id = b.link_to_event WHERE b.category = 'Speaker Gifts' ), total_spent AS ( SELECT SUM(spent) AS total_amount FROM budget WHERE category = 'Speaker Gifts' ) SELECT t.total_amount, ed.event_name FROM total_spent t CROSS JOIN event_details ed;




WITH g_cols AS ( SELECT gs.GasStationID, SUM(t.Amount) AS TotalRevenue FROM gasstations gs JOIN transactions_1k t ON gs.GasStationID = t.GasStationID GROUP BY gs.GasStationID ), max_gas_station AS ( SELECT GasStationID FROM g_cols WHERE TotalRevenue = (SELECT MAX(TotalRevenue) FROM g_cols) ) SELECT gs.GasStationID, gs.ChainID, gs.Country, gs.Segment FROM gasstations gs JOIN max_gas_station mgs ON gs.GasStationID = mgs.GasStationID;
WITH p_price AS ( SELECT CustomerID, Price / Amount AS price_per_unit FROM transactions_1k WHERE ProductID = 5 ), p_filtered AS ( SELECT CustomerID FROM p_price WHERE price_per_unit > 29.00 ), ym_data AS ( SELECT CustomerID, Consumption FROM yearmonth WHERE Date = '201208' ) SELECT ym.CustomerID, ym.Consumption FROM ym_data ym JOIN p_filtered pf ON ym.CustomerID = pf.CustomerID;
