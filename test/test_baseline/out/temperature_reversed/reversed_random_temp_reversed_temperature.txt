SELECT T3.Description, T4.Consumption FROM transactions_1k AS T1 INNER JOIN products AS T3 ON T1.ProductID = T3.ProductID INNER JOIN yearmonth AS T4 ON T1.CustomerID = T4.CustomerID WHERE T1.Date LIKE '201208%' AND T1.ProductID = 5 AND T1.Price / T1.Amount > 29.00
