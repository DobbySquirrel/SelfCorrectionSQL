WITH c_cols AS ( SELECT a.molecule_id FROM atom AS a WHERE a.element = 's' ), c_bonds AS ( SELECT b.molecule_id FROM c_cols INNER JOIN bond AS b ON c_cols.molecule_id = b.molecule_id ), c_double_bonds AS ( SELECT b.molecule_id FROM c_bonds INNER JOIN bond AS b ON c_bonds.molecule_id = b.molecule_id WHERE b.bond_type = '=' ), t_final AS ( SELECT m.label FROM c_double_bonds INNER JOIN molecule AS m ON c_double_bonds.molecule_id = m.molecule_id ) SELECT * FROM t_final;
