WITH c_cols AS ( SELECT ss.cds, ss.`AvgScrMath`, f.`Charter Funding Type` FROM satscores AS ss INNER JOIN frpm AS f ON ss.cds = f.`CDSCode` INNER JOIN schools AS sc ON ss.cds = sc.`CDSCode` WHERE sc.`County` = 'Riverside' ), c_rows AS ( SELECT cds, `AvgScrMath` FROM c_cols WHERE `AvgScrMath` IS NOT NULL AND `AvgScrMath` > 400 ), t_join AS ( SELECT r.cds, f.`Charter Funding Type` FROM c_rows r INNER JOIN frpm f ON r.cds = f.`CDSCode` ) SELECT `Charter Funding Type` FROM t_join;
WITH c_cols AS ( SELECT ss.cds, ss.`AvgScrRead`, s.`Phone` FROM satscores AS ss INNER JOIN schools AS s ON ss.cds = s.`CDSCode` WHERE s.`District` = 'Fresno Unified' ), c_rows AS ( SELECT cds, `AvgScrRead` FROM c_cols WHERE `AvgScrRead` IS NOT NULL ), t_join AS ( SELECT r.cds, r.`AvgScrRead`, s.`Phone` FROM c_rows r INNER JOIN schools s ON r.cds = s.`CDSCode` ), t_agg AS ( SELECT cds, MIN(`AvgScrRead`) AS min_avg_scr_read FROM t_join GROUP BY cds ), t_final AS ( SELECT t.cds, t.`Phone` FROM t_agg a INNER JOIN t_join t ON a.cds = t.cds WHERE a.min_avg_scr_read = ( SELECT MIN(min_avg_scr_read) FROM t_agg ) ) SELECT `Phone` FROM t_final;
WITH c_cols AS ( SELECT c.client_id, c.gender, c.district_id, d.A11 FROM client c JOIN district d ON c.district_id = d.district_id ), c_rows AS ( SELECT client_id, gender, district_id, A11 FROM c_cols WHERE gender = 'M' AND A11 > 8000 ), t_join AS ( SELECT c.client_id, c.district_id, d.A3 FROM c_rows c JOIN district d ON c.district_id = d.district_id ), c_fuzzy AS ( SELECT client_id FROM t_join WHERE A3 LIKE '%North Bohemia%' OR A3 LIKE 'North Bohemia%' OR A3 LIKE '%North% Bohemia%' OR A3 LIKE 'North% Bohemia%' OR A3 LIKE '%North% Bohemia%' ), c_final AS ( SELECT COUNT(*) AS answer FROM c_fuzzy ) SELECT answer FROM c_final;
